# LỆNH #11 — Cloud Functions PUSH (FCM) đi kèm zip v119-43 (04/09/2026)

Trước khi chạy: (1) Firebase console → Project settings → **Cloud Messaging** → Web Push certificates → **Generate key pair** → copy **Key pair (public)** gửi em (khoá công khai, không phải secret) để em nhét `VAPID_KEY` vào `firebase-config.js` của zip kế. (2) Chạy LỆNH dưới.

```bash
gcloud config set project smartlead-z15 >/dev/null 2>&1
gcloud services enable fcm.googleapis.com firebasecloudmessaging.googleapis.com eventarc.googleapis.com 2>&1 | tail -2
cd ~/firebase-s13/functions && cat > push.js <<'EOF'
/* v119-43: PUSH FCM (web push qua service worker sw.js của app)
   (1) lead chuyển 'responded' (khách phản hồi) → người phụ trách, không có thì admin của brand
   (2) lead NÓNG mới tạo → admin của brand
   (3) mỗi 15' nhắc hẹn fu_at tới hạn → người phụ trách
   Gửi DATA-ONLY (không có "notification") để SW tự hiện → không bị double. Token chết tự dọn khỏi users.fcmTokens. */
import { onDocumentWritten } from 'firebase-functions/v2/firestore';
import { onSchedule } from 'firebase-functions/v2/scheduler';
import { getApps, initializeApp } from 'firebase-admin/app';
import { getFirestore, FieldValue } from 'firebase-admin/firestore';
import { getMessaging } from 'firebase-admin/messaging';
if (!getApps().length) initializeApp();
const REGION = 'asia-southeast1', SITE = 'https://smartlead.z15miracle.com.vn/app.html';
const db = () => getFirestore();
async function tokensFor(uids) {
  const out = [];
  for (const uid of [...new Set((uids || []).filter(Boolean))]) {
    const d = (await db().doc('users/' + uid).get()).data() || {};
    if (d.active === false) continue;
    (Array.isArray(d.fcmTokens) ? d.fcmTokens : []).forEach(t => out.push({ uid, t }));
  }
  return out;
}
async function brandAdmins(brand) {
  if (!brand) return [];
  const q = await db().collection('users').where('brand', '==', brand).get(); // lọc role phía code → không cần composite index
  return q.docs.filter(d => { const x = d.data() || {}; return x.role === 'admin' && x.active !== false; }).map(d => d.id);
}
async function send(recips, data) {
  if (!recips.length) return 0;
  const payload = Object.fromEntries(Object.entries(data).map(([k, v]) => [k, String(v == null ? '' : v)]));
  const res = await getMessaging().sendEachForMulticast({ tokens: recips.map(r => r.t), data: payload, webpush: { headers: { Urgency: 'high', TTL: '3600' } } });
  await Promise.all(res.responses.map(async (r, i) => {
    if (r.success) return;
    const code = String((r.error && r.error.code) || '');
    if (/registration-token-not-registered|invalid-registration-token|invalid-argument/.test(code)) {
      const rc = recips[i]; try { await db().doc('users/' + rc.uid).update({ fcmTokens: FieldValue.arrayRemove(rc.t) }); } catch (_) { }
    }
  }));
  return res.successCount;
}
export const pushOnLead = onDocumentWritten({ region: REGION, document: 'leads/{id}', retry: false }, async (ev) => {
  const before = ev.data && ev.data.before.exists ? ev.data.before.data() : null;
  const after = ev.data && ev.data.after.exists ? ev.data.after.data() : null;
  if (!after) return;
  const id = ev.params.id, name = after.name || 'Lead', brand = after.brand || '';
  if (after.stage === 'responded' && (!before || before.stage !== 'responded')) {
    const uids = after.assignee_uid ? [after.assignee_uid] : await brandAdmins(brand);
    const n = await send(await tokensFor(uids), { title: '💬 Khách đã phản hồi: ' + name, body: String(after.need || after.intent || 'Mở Phản hồi khách để trả lời ngay').slice(0, 120), link: SITE + '#replies', tag: 'reply-' + id });
    console.log('push reply', id, 'sent', n); return;
  }
  if (!before && after.temp === 'hot') {
    const n = await send(await tokensFor(await brandAdmins(brand)), { title: '🔥 Lead nóng mới' + (after.score ? ' (' + after.score + 'đ)' : '') + ': ' + name, body: String(after.need || after.intent || '').slice(0, 120), link: SITE + '#feed', tag: 'hot-' + id });
    console.log('push hot', id, 'sent', n);
  }
});
export const pushDueFollowups = onSchedule({ region: REGION, schedule: 'every 15 minutes', timeZone: 'Asia/Ho_Chi_Minh', retryCount: 0 }, async () => {
  const now = Date.now(), from = now - 15 * 60000;
  const q = await db().collection('leads').where('fu_at', '>', from).where('fu_at', '<=', now).get();
  let sent = 0;
  for (const d of q.docs) {
    const l = d.data() || {};
    if (l.stage === 'closed' || l.lost || l.dropped || !l.assignee_uid) continue;
    sent += await send(await tokensFor([l.assignee_uid]), { title: '⏰ Đến giờ chăm: ' + (l.name || 'Lead'), body: String(l.fu_note || 'Hẹn chăm lead tới hạn').slice(0, 120), link: SITE + '#tasks', tag: 'fu-' + d.id });
  }
  console.log('push due', q.size, 'sent', sent);
});
EOF
grep -q "push.js" index.js || echo "export * from './push.js';" >> index.js
node --check push.js && echo "SYNTAX OK" && grep -n "push.js" index.js
cd ~/firebase-s13 && firebase deploy --only functions:pushOnLead,functions:pushDueFollowups 2>&1 | tail -8
gcloud functions describe pushOnLead --region=asia-southeast1 --gen2 --format="value(state,eventTrigger.eventType)" 2>/dev/null
```
Kỳ vọng: `SYNTAX OK`, deploy `Deploy complete!`, describe `ACTIVE google.cloud.firestore.document.v1.written`.
Lưu ý: trigger Firestore v2 cần vùng function khớp vị trí database; nếu deploy báo lỗi vùng, gửi em output để đổi `REGION`.
