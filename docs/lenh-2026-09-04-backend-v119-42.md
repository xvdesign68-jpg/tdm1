## LỆNH #10 — Cloud Function `setUserLock` (khoá THẬT tài khoản: Auth disable + thu hồi phiên)
```bash
cd ~/firebase-s13/functions && cat > admin.js <<'EOF'
/* v119-42: setUserLock - khoá/mở THẬT tài khoản: vô hiệu hoá Firebase Auth + thu hồi refresh token + cờ users.active/authDisabled. Chỉ Super Admin gọi. */
import { onRequest } from 'firebase-functions/v2/https';
import { getApps, initializeApp } from 'firebase-admin/app';
import { getAuth } from 'firebase-admin/auth';
import { getFirestore, FieldValue } from 'firebase-admin/firestore';
if (!getApps().length) initializeApp();
const SUPER_EMAIL = (process.env.SUPER_EMAIL || 'xuanvinhsc68.work@gmail.com').toLowerCase();
async function verifySuper(req) {
  const m = String(req.headers.authorization || '').match(/^Bearer (.+)$/); if (!m) return null;
  const dec = await getAuth().verifyIdToken(m[1]); const em = String(dec.email || '').toLowerCase();
  if (em === SUPER_EMAIL) return dec;
  const d = ((await getFirestore().doc('users/' + dec.uid).get()).data()) || {};
  return (d.role === 'superadmin' && d.active !== false) ? dec : null;
}
export const setUserLock = onRequest({ region: 'asia-southeast1', cors: true, timeoutSeconds: 30 }, async (req, res) => {
  if (req.method !== 'POST') return res.status(405).json({ error: 'method' });
  let caller; try { caller = await verifySuper(req); } catch (e) { return res.status(401).json({ error: 'unauthenticated' }); }
  if (!caller) return res.status(403).json({ error: 'forbidden' });
  const uid = String((req.body && req.body.uid) || ''); const disabled = !!(req.body && req.body.disabled);
  if (!uid) return res.status(400).json({ error: 'missing_uid' });
  if (uid === caller.uid) return res.status(400).json({ error: 'cannot_lock_self' });
  const target = await getAuth().getUser(uid).catch(() => null);
  if (!target) return res.status(404).json({ error: 'user_not_found' });
  if (String(target.email || '').toLowerCase() === SUPER_EMAIL) return res.status(400).json({ error: 'cannot_lock_super' });
  await getAuth().updateUser(uid, { disabled });
  if (disabled) await getAuth().revokeRefreshTokens(uid);
  await getFirestore().doc('users/' + uid).set({ active: !disabled, authDisabled: disabled, lockedAt: disabled ? Date.now() : FieldValue.delete(), lockedBy: disabled ? (caller.email || caller.uid) : FieldValue.delete(), updatedAt: FieldValue.serverTimestamp() }, { merge: true });
  return res.json({ ok: true, uid, disabled });
});
EOF
grep -q "admin.js" index.js || echo "export * from './admin.js';" >> index.js
node --check admin.js && echo "SYNTAX OK" && grep -n "admin.js" index.js
cd ~/firebase-s13 && firebase deploy --only functions:setUserLock 2>&1 | tail -6
curl -sS -X POST "https://asia-southeast1-smartlead-z15.cloudfunctions.net/setUserLock" -H "Content-Type: application/json" -d '{}'; echo
```
Kỳ vọng dòng cuối: `{"error":"unauthenticated"}` (chưa có token = từ chối đúng).
